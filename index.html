function shuffleAnswers(question) { const answers = question.answers.map((answer, index) => ({ text: answer, correct: index === question.correctIndex })); // خلط عشوائي للخيارات for (let i = answers.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [answers[i], answers[j]] = [answers[j], answers[i]]; } return { ...question, answers: answers.map(answer => answer.text), correctIndex: answers.findIndex(answer => answer.correct) }; } // مثال على سؤال const question = { question: "ما أعمق بحيرة في العالم؟", answers: [ "بحيرة فيكتوريا", "بحيرة بايكال", "بحيرة سوبيريور", "بحيرة تنجانيقا" ], correctIndex: 1 }; const randomizedQuestion = shuffleAnswers(question); console.log(randomizedQuestion);
